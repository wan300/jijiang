## 3. 请求封装

### 3.1 `request.ts` 核心实现

```typescript
import { useUserStore } from '@/store/user';
import { toast } from '@/utils/toast';

const BASE_URL = import.meta.env.VITE_API_BASE || 'https://api.jijiang.com';

type Method = 'GET' | 'POST' | 'PUT' | 'DELETE';

interface RequestOptions<T = unknown> {
  url: string;
  method?: Method;
  data?: Record<string, unknown>;
  params?: Record<string, unknown>;
  header?: Record<string, string>;
  skipAuth?: boolean;
  skipToast?: boolean;
  retry?: number;
}

export interface ApiResult<T> {
  code: number;
  message: string;
  data: T;
  timestamp: number;
}

export async function request<T = unknown>(options: RequestOptions): Promise<T> {
  const user = useUserStore();
  const header: Record<string, string> = {
    'Content-Type': 'application/json',
    ...options.header,
  };
  if (!options.skipAuth && user.token) {
    header.Authorization = `Bearer ${user.token}`;
  }

  let url = BASE_URL + options.url;
  if (options.params) {
    const qs = Object.entries(options.params)
      .filter(([, v]) => v !== undefined && v !== null)
      .map(([k, v]) => `${k}=${encodeURIComponent(String(v))}`)
      .join('&');
    if (qs) url += (url.includes('?') ? '&' : '?') + qs;
  }

  try {
    const res = await uni.request({
      url,
      method: options.method || 'GET',
      data: options.data,
      header,
      timeout: 15000,
    });
    const body = res.data as ApiResult<T>;
    if (body.code === 0) return body.data;
    if (body.code === 401) {
      await user.clearAndRedirect();
      throw new Error('登录已过期');
    }
    if (!options.skipToast) toast(body.message || '请求失败');
    throw new BizError(body.code, body.message);
  } catch (e: any) {
    if (!options.skipToast && !(e instanceof BizError)) toast('网络异常，请稍后重试');
    throw e;
  }
}

export class BizError extends Error {
  constructor(public code: number, message: string) {
    super(message);
  }
}
```

## 4. 状态管理（Pinia）

### 4.1 用户登录态

```typescript
import { defineStore } from 'pinia';
import { loginByCode } from '@/api/auth';

interface UserState {
  token: string;
  refreshToken: string;
  userInfo: UserInfo | null;
  currentRole: 1 | 2;
}

export const useUserStore = defineStore('user', {
  state: (): UserState => ({
    token: uni.getStorageSync('token') || '',
    refreshToken: uni.getStorageSync('refreshToken') || '',
    userInfo: uni.getStorageSync('userInfo') || null,
    currentRole: uni.getStorageSync('currentRole') || 1,
  }),
  getters: {
    isLogin: (s) => !!s.token,
    isVerified: (s) => s.userInfo?.verifyStatus === 2,
    isSeller: (s) => s.currentRole === 2,
  },
  actions: {
    async login() {
      const { code } = await uni.login({ provider: 'weixin' });
      const res = await loginByCode(code);
      this.setTokens(res.accessToken, res.refreshToken);
      this.userInfo = res.userInfo;
      uni.setStorageSync('userInfo', res.userInfo);
    },
    setTokens(access: string, refresh: string) {
      this.token = access;
      this.refreshToken = refresh;
      uni.setStorageSync('token', access);
      uni.setStorageSync('refreshToken', refresh);
    },
    switchRole(role: 1 | 2) {
      this.currentRole = role;
      uni.setStorageSync('currentRole', role);
      uni.$emit('role:changed', role);
    },
    async clearAndRedirect() {
      this.$reset();
      uni.clearStorageSync();
      uni.reLaunch({ url: '/pages/login/index' });
    },
  },
});
```

## 5. 路由守卫 / 页面保护

小程序原生没有路由守卫概念，通过以下方式实现：

### 5.1 在 App.vue 统一拦截

```vue
<script setup lang="ts">
import { onLaunch } from '@dcloudio/uni-app';
import { useUserStore } from '@/store/user';

onLaunch(async () => {
  const user = useUserStore();
  if (user.token) {
    try {
      await user.refreshUserInfo();
    } catch { user.clearAndRedirect(); }
  }
});
</script>
```

### 5.2 在需要认证的页面 `onLoad`

```typescript
import { useUserStore } from '@/store/user';

onLoad(() => {
  const user = useUserStore();
  if (!user.isLogin) {
    uni.redirectTo({ url: `/pages/login/index?redirect=${encodeURIComponent(getCurrentPageUrl())}` });
    return;
  }
  if (!user.isVerified) {
    uni.showModal({
      title: '需要实名认证',
      content: '请先完成学生身份认证',
      success: (res) => res.confirm && uni.navigateTo({ url: '/pages/user/verify' }),
    });
  }
});
```

### 6.3 下单页（`pages/order/create.vue`）

V1.0 使用 **虎皮椒个人支付 + 独立 B 支付服务**。小程序端不直接调用微信商户 `uni.requestPayment`，也不接触虎皮椒参数。前端只负责创建订单、调用 A 后端支付入口、打开返回的收银台/二维码，并轮询 A 后端订单状态。

支付链路：

- 小程序 → A 后端：`POST /api/payment/create`
- A 后端 → B 支付服务：`POST /internal/payment/orders`（HMAC-SHA256）
- B 支付服务 → 虎皮椒：创建真实支付单
- 虎皮椒 → B 支付服务：`POST /api/payment/xunhu-notify`
- B 支付服务 → A 后端：`POST /internal/payment/callback`

前端支付接口建议返回：

```typescript
export interface PaymentCreateResult {
  channel: 'XUNHUPAY';
  orderId: number;
  orderNo: string;
  tradeOrderId: string;
  payUrl: string;
  qrCodeUrl?: string;
  expireSeconds: number;
}

export function payOrder(orderId: number) {
  return request<PaymentCreateResult>({
    url: '/api/payment/create',
    method: 'POST',
    data: { orderId },
  });
}
```

import { ref } from 'vue';
import { createOrder } from '@/api/order';

const paying = ref(false);

async function submit() {
  if (paying.value) return;
  paying.value = true;
  try {
    const order = await createOrder({ serviceId, pricePlanKey, quantity, remark });
    const cashier = await payOrder(order.id);

    // 虎皮椒个人支付返回的是付款链接/二维码地址，不是微信 JSAPI 支付参数。
    // 支付结果页统一承载收银台入口和状态轮询。
    const payUrl = encodeURIComponent(cashier.payUrl);
    uni.redirectTo({
      url: `/pages/order/pay-result?orderNo=${order.orderNo}&payUrl=${payUrl}`,
    });
  } finally {
    paying.value = false;
  }
}

`/pages/order/pay-result` 不依赖前端跳转判断成功，而是持续轮询后端订单状态：

```vue
<template>
  <view class="pay-result">
    <view v-if="!paid">
      <button @click="openCashier">打开收银台</button>
      <image v-if="payUrl" :src="payUrl" mode="widthFix" />
      <text>支付完成后请返回本页，系统会自动确认订单状态</text>
    </view>
  </view>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import { onLoad, onUnload } from '@dcloudio/uni-app';
import { getOrderDetail } from '@/api/order';

const orderNo = ref('');
const payUrl = ref('');
const paid = ref(false);
let timer: ReturnType<typeof setInterval> | undefined;

onLoad((query) => {
  orderNo.value = String(query?.orderNo || '');
  payUrl.value = decodeURIComponent(String(query?.payUrl || ''));
  timer = setInterval(checkPaid, 2000);
  checkPaid();
});

onUnload(() => {
  if (timer) clearInterval(timer);
});

function openCashier() {
  uni.navigateTo({
    url: `/pages/common/webview?url=${encodeURIComponent(payUrl.value)}&orderNo=${orderNo.value}&from=pay`,
  });
}

async function checkPaid() {
  if (!orderNo.value) return;
  const order = await getOrderDetail(orderNo.value);
  if (order.status >= 20) {
    paid.value = true;
    if (timer) clearInterval(timer);
    uni.redirectTo({ url: `/pages/order/detail?orderNo=${orderNo.value}` });
  }
}
</script>
```

> **注意**：前端展示“待确认支付结果”即可，最终状态只信任 B 支付服务完成虎皮椒回调验签后回推 A 写入的订单状态，避免用户伪造 `return_url` 或手动关闭收银台导致误判。

- 列表：虚拟列表（`uv-virtual-list`）或分页加载

### 8.2 图片上传

```typescript
async function chooseAndUpload() {
  const [err, res] = await uni.chooseImage({ count: 1, sizeType: ['compressed'] });
  if (err) return;
  const tmp = res.tempFilePaths[0];

  const info = await uni.getImageInfo({ src: tmp });
  if (info.width * info.height > 1920 * 1080) {
    const { tempFilePath } = await uni.compressImage({ src: tmp, quality: 70 });
    return uploadToOss(tempFilePath);
  }
  return uploadToOss(tmp);
}
```

## 11. 安全约束

- **禁止**在客户端存储敏感字段（如未脱敏的手机号）
- **Token**：仅存于 `uni.setStorageSync`；不与业务参数混传
- **防跳单**：站内信发送前正则拦截"微信、QQ、数字串"（详见 [07-安全](./07-安全与风控设计.md)）
- **请求签名**（可选）：V2.0 后可增加请求签名防篡改
- **开发调试**：`console.log` 生产环境自动禁用（Vite 插件）
